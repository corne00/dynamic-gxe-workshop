8 · Build your own model

Choose the drivers, the library and the fitting method, and see the resulting equation and its prediction for years that were left out of the fit. The calculations run in the browser and follow the functions of the previous steps: the genotype is fitted to all years for the coefficients, and leave-one-year-out validation gives the held-out error.

d3 = window.d3
Plot = window.Plot
builder = FileAttachment("data/interactive/builder.json").json()
palette = ["#014E81", "#00A6D6", "#8B1E3F", "#469C2A", "#6F1D77", "#EF60A3", "#00A390", "#6B7280"]
viewof chosen = toggles("Drivers", builder.drivers, builder.drivers.map(() => "#014E81"), builder.drivers)
viewof degree = buttons("Powers up to degree", [1, 2, 3], 3)
viewof products = buttons("Products of pairs", ["no", "yes"], "no")
examples = [
  {label: "heat", line: "heat = max(Tmax - 30, 0)"},
  {label: "frost", line: "frost = max(2 - Tmin, 0)"},
  {label: "optimum", line: "opt = exp(-(Tmean - 20)^2 / (2 * 5^2))"},
  {label: "log rain", line: "lograin = log(1 + Rain)"},
  {label: "rain, 7 days", line: "rain7 = roll(Rain, 7)"},
  {label: "cumulative RAD", line: "cumRAD = cum(RAD)"},
  {label: "vapour deficit", line: "vpd = VPD"},
  {label: "RAD / DL", line: "ratio = RAD / DL"},
  {label: "temperature × water", line: "TW = Tmean * RAW"},
  {label: "lagged Tmax", line: "lag3 = lag(Tmax, 3)"},
  {label: "yearly cycle", line: "cycle = sin(2 * pi * das / 365)"}]
viewof extraText = expressions("Extra terms", examples)
parsed = Builder.parseExtras(extraText)
extras = parsed.terms
html`${parsed.errors.map(e => html`<p class="interactive-note">Line ${e.line}: ${e.message}</p>`)}`

Extra terms are functions of the raw weather. Each line adds one term to the library, standardised like the drivers: write name = expression, or click an example. The variables are Tmin, Tmax, Rain, RAD, VP, DL, TT, RAW, Tmean, VPD (vapour pressure deficit, from Tmax and VP) and das (days after sowing). The operators are + - * / ^, and the functions are exp, log, sqrt, abs, sin, cos, tan, tanh, min and max (elementwise), roll(x, k) (mean of the last k days), cum(x) and lag(x, k).

library = Builder.termSpecs(chosen, degree, products === "yes")
K = library.names.length + extras.length
catalogue = Builder.methods(K)
available = Object.keys(catalogue).filter(m => catalogue[m].values.length > 0)
viewof method = select("Method", available, "STLSQ")
spec = catalogue[method]
viewof step = spec.param
  ? slider(spec.param, spec.values, Math.min(Math.floor(spec.values.length / 2), spec.values.length - 1), v => (+v).toPrecision(2))
  : Object.assign(html`<div></div>`, {value: 0})
viewof genotype = select("Genotype", builder.genotypes, builder.genotypes[0])
viewof year = slider("Year", builder.years, builder.years.indexOf(1998))
result = {
  try {
    return Builder.run(builder, builder.genotypes.indexOf(genotype), {
      drivers: chosen, degree, pairs: products === "yes", extras, method, value: spec.values[step]});
  } catch (e) {
    return {error: e.message};
  }
}
result.error ? html`<p class="interactive-note">${result.error}</p>` : html`<p class="interactive-note">${K} terms in the library, ${result.nterms} retained. Held-out RMSE ${result.rmse.toFixed(3)} t/ha over all years; at maturity ${result.maturityRmse.toFixed(3)} t/ha. (${result.ms.toFixed(0)} ms)</p>`
{
  if (result.error) return html``;
  const kept = result.terms.map((term, i) => ({term, effect: result.scaled[i]})).filter(d => d.effect !== 0);
  const reach = d3.max(kept, d => Math.abs(d.effect)) || 1;
  const coefficients = Plot.plot({
    width: 520, height: 40 + 16 * Math.max(kept.length, 1), marginLeft: 70,
    x: {domain: [-reach, reach], label: "scaled coefficient"},
    y: {domain: kept.map(d => d.term), label: null},
    marks: [Plot.barX(kept, {x: "effect", y: "term", fill: d => d.effect > 0 ? "#014E81" : "#8B1E3F"}),
            Plot.ruleX([0])]
  });
  const g = builder.genotypes.indexOf(genotype), s = year, shown = builder.years[s];
  const das = i => builder.das0 + i;
  const observed = builder.Y[g][s].map((y, i) => ({das: das(i), y}));
  const predicted = result.predicted[s].map((y, i) => ({das: das(i), y}));
  const trajectory = Plot.plot({
    width: 520, height: 260, marginLeft: 50,
    x: {label: "days after sowing"}, y: {label: "biomass (t/ha)", domain: [0, d3.max([...observed, ...predicted], d => d.y) * 1.05]},
    marks: [Plot.line(observed, {x: "das", y: "y", stroke: "#9CA3AF", strokeWidth: 4}),
            Plot.line(predicted, {x: "das", y: "y", stroke: "#8B1E3F", strokeWidth: 2})]
  });
  const finals = builder.years.map((yr, i) => ({
    year: yr, observed: builder.Y[g][i].at(-1), predicted: result.predicted[i].at(-1)}));
  const [lo, hi] = d3.extent(finals.flatMap(d => [d.observed, d.predicted]));
  const scatter = Plot.plot({
    width: 520, height: 300, marginLeft: 50,
    x: {domain: [lo - 0.5, hi + 0.5], label: "APSIM biomass at maturity (t/ha)"},
    y: {domain: [lo - 0.5, hi + 0.5], label: "predicted (t/ha)"},
    color: {type: "linear", scheme: "blues"},
    marks: [Plot.line([[lo - 0.5, lo - 0.5], [hi + 0.5, hi + 0.5]], {stroke: "#9CA3AF"}),
            Plot.dot(finals, {x: "observed", y: "predicted", fill: "year", stroke: "#014E81", r: 4}),
            Plot.dot(finals.filter(d => d.year === shown), {x: "observed", y: "predicted", fill: "#8B1E3F", stroke: "#8B1E3F", r: 6})]
  });
  return html`<div>${coefficients}${trajectory}${scatter}</div>
    <p class="interactive-note">Grey: APSIM. Red: prediction for year ${shown}, from a model fitted to the other years. Each point in the lower panel is one year.</p>`;
}

The same model in R and Python

The choices above correspond to the following calls, with wx and traj loaded as in step 2.

code = {
  const f = v => (v === null ? "" : `${+(+v).toPrecision(3)}`);
  const call = {
    OLS: ["fit_ols", () => "", () => ""],
    ridge: ["fit_ridge", v => `, lambda = ${f(v)}`, v => `, lam=${f(v)}`],
    lasso: ["fit_lasso", v => `, lambda = ${f(v)}`, v => `, lam=${f(v)}`],
    "elastic net": ["fit_enet", v => `, lambda = ${f(v)}`, v => `, lam=${f(v)}`],
    "best subset": ["fit_subset", v => `, k = ${f(v)}`, v => `, k=${f(v)}`],
    backward: ["fit_subset", v => `, k = ${f(v)}, method = "backward"`, v => `, k=${f(v)}, method="backward"`],
    STLSQ: ["fit_stlsq", v => `, threshold = ${f(v)}`, v => `, threshold=${f(v)}`]
  }[method];
  const v = spec.values[step];
  const rArgs = call[1](v), pyArgs = call[2](v);
  const rDrivers = `c(${chosen.map(d => `"${d}"`).join(", ")})`;
  const pyDrivers = `[${chosen.map(d => `"${d}"`).join(", ")}]`;
  const pairs = products === "yes";
  const rExtra = extras.length ? `
extra <- extra_terms(wx, c(${extras.map(e => `${e.name} = "${e.expr}"`).join(",\n                           ")}))
Theta <- bind_terms(Theta, extra)` : "";
  const pyExtra = extras.length ? `
extra = extra_terms(wx, {${extras.map(e => `"${e.name}": "${e.expr}"`).join(",\n                         ")}})
Theta = bind_terms(Theta, extra)` : "";
  return {
    r: `drv <- make_drivers(wx, ${rDrivers})
Theta <- make_library(drv, degree = ${degree}, pairs = ${pairs ? "TRUE" : "FALSE"})${rExtra}
tr <- make_trajectories(traj, drv)
fit <- function(X, dy) ${call[0]}(X, dy${rArgs})

d <- make_design(Theta, tr, "${genotype}")
coef <- fit(d$X, d$dy)
P <- cross_validate(Theta, tr, fit, "${genotype}")
evaluate(P, tr)`,
    py: `drv = make_drivers(wx, ${pyDrivers})
Theta = make_library(drv, degree=${degree}, pairs=${pairs ? "True" : "False"})${pyExtra}
tr = make_trajectories(traj, drv)

def fit(X, dy):
    return ${call[0]}(X, dy${pyArgs})

d = make_design(Theta, tr, "${genotype}")
coef = fit(d.X, d.dy)
P = cross_validate(Theta, tr, fit, ["${genotype}"])
print(evaluate(P, tr))`};
}
